class Solution:
    def checkString(self, s):
        
        # Initializing vowel count
        # and consonant count to 0
        v = 0
        c = 0
        
        # code here
        vowels=['a','e','i','o','u']
        l=len(s)
        if s.isalpha():
            for i in range(l):
                if s[i] in vowels:
                    v+=1
                else:
                    c+=1
                    
        if v>c:
            print("Yes")
        elif c>v:
            print("No")
        else:
            print("Same")
